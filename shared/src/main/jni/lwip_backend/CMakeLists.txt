cmake_minimum_required(VERSION 3.15)
project(diplay_lwip_host C)
if(NOT WIN32)
  find_package(Threads REQUIRED)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(DIPLAY_SANITIZERS "Enable address and undefined-behavior sanitizers" OFF)
if(DIPLAY_SANITIZERS)
  if(WIN32 OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require a supported Linux compiler")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  # lwIP pools remain one allocation from ASan's perspective. Guard individual pool/heap
  # allocations as well so an intra-pool overwrite is not hidden by that allocator model.
  add_compile_definitions(MEM_OVERFLOW_CHECK=2 MEMP_OVERFLOW_CHECK=2 MEM_SANITY_CHECK=1 MEMP_SANITY_CHECK=1)
endif()
file(GLOB LWIP_CORE CONFIGURE_DEPENDS third_party/lwip/src/core/*.c third_party/lwip/src/core/ipv6/*.c)
add_library(lwip_stack STATIC ${LWIP_CORE} third_party/lwip/src/netif/ethernet.c)
target_include_directories(lwip_stack PUBLIC config port third_party/lwip/src/include)
add_library(diplay_core STATIC src/core.c src/netif.c src/prototype.c src/prototype_echo.c src/endpoint_registry.c src/udp_endpoint.c src/tcp_endpoint.c)
target_compile_definitions(diplay_core PRIVATE DP_PROTO_TESTING=1)
target_include_directories(diplay_core PUBLIC include PRIVATE src)
target_link_libraries(diplay_core PUBLIC lwip_stack)
target_compile_options(diplay_core PRIVATE -Wall -Wextra -Werror)
enable_testing()
add_executable(endpoint_registry_tests tests/endpoint_registry_tests.c)
target_include_directories(endpoint_registry_tests PRIVATE src)
target_link_libraries(endpoint_registry_tests PRIVATE diplay_core)
add_test(NAME endpoint_registry COMMAND endpoint_registry_tests)
set_tests_properties(endpoint_registry PROPERTIES TIMEOUT 15)
add_executable(ndp_option_tests tests/ndp_option_tests.c)
target_link_libraries(ndp_option_tests PRIVATE diplay_core)
add_test(NAME ndp_option_bounds COMMAND ndp_option_tests)
set_tests_properties(ndp_option_bounds PROPERTIES TIMEOUT 15)
add_executable(reassembly_tests tests/reassembly_tests.c)
target_link_libraries(reassembly_tests PRIVATE diplay_core)
add_test(NAME reassembly_empty_context COMMAND reassembly_tests)
set_tests_properties(reassembly_empty_context PROPERTIES TIMEOUT 15)
add_executable(tcp_option_tests tests/tcp_option_tests.c)
target_link_libraries(tcp_option_tests PRIVATE diplay_core)
add_test(NAME tcp_option_bounds COMMAND tcp_option_tests)
set_tests_properties(tcp_option_bounds PROPERTIES TIMEOUT 15)
add_library(wire_faults STATIC tests/wire_faults.c)
target_include_directories(wire_faults PUBLIC tests)
target_compile_options(wire_faults PRIVATE -Wall -Wextra -Werror)
add_executable(wire_fault_tests tests/wire_fault_tests.c)
target_link_libraries(wire_fault_tests PRIVATE wire_faults)
target_compile_options(wire_fault_tests PRIVATE -Wall -Wextra -Werror)
foreach(scenario capacity delay drop reorder duplicate validation)
  add_test(NAME wire_${scenario} COMMAND wire_fault_tests ${scenario})
  set_tests_properties(wire_${scenario} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(prototype_tests tests/prototype_tests.c)
target_compile_definitions(prototype_tests PRIVATE DP_PROTO_TESTING=1)
target_link_libraries(prototype_tests PRIVATE diplay_core)
if(NOT WIN32)
  target_link_libraries(prototype_tests PRIVATE Threads::Threads)
endif()
target_compile_options(prototype_tests PRIVATE -Wall -Wextra -Werror)
foreach(scenario lifecycle queue stack entropy thread generation)
  add_test(NAME prototype_${scenario} COMMAND prototype_tests ${scenario})
  set_tests_properties(prototype_${scenario} PROPERTIES TIMEOUT 15)
endforeach()
add_executable(frame_tests tests/frame_tests.c tests/echo_services.c)
target_link_libraries(frame_tests PRIVATE diplay_core)
if(NOT WIN32)
  target_link_libraries(frame_tests PRIVATE Threads::Threads)
endif()
target_compile_options(frame_tests PRIVATE -Wall -Wextra -Werror)
foreach(scenario ndp ndp_options dad tcp tcp_large_refused tcp_endpoint tcp_backpressure tcp_endpoint_limits tcp_endpoint_lifecycle tcp_write tcp_half_close tcp_write_io tcp_shutdown_pressure tcp_graceful_close tcp_graceful_pressure tcp_graceful_failure tcp_liveness tcp_reuse tcp_connect tcp_connect_cancel tcp_connect_limits tcp_connect_retry udp udp_endpoint udp_endpoint_fragments pmtu lifecycle not_ready tx_not_ready tx_failed malformed fragments api thread)
  add_test(NAME ${scenario} COMMAND frame_tests ${scenario})
  set_tests_properties(${scenario} PROPERTIES TIMEOUT 15)
endforeach()
if(CMAKE_SYSTEM_NAME STREQUAL "Linux" OR ANDROID)
  add_executable(lwip_kernel_echo tests/kernel_echo.c tests/echo_services.c tests/endpoint_services.c)
  target_link_libraries(lwip_kernel_echo PRIVATE diplay_core wire_faults)
  target_compile_options(lwip_kernel_echo PRIVATE -Wall -Wextra -Werror)
  add_executable(lwip_namespace_exec tests/namespace_exec.c)
  add_executable(lwip_kernel_peer tests/kernel_peer.c)
  add_executable(lwip_kernel_offloads tests/kernel_offloads.c)
  target_compile_options(lwip_namespace_exec PRIVATE -Wall -Wextra -Werror)
  target_compile_options(lwip_kernel_peer PRIVATE -Wall -Wextra -Werror)
  target_compile_options(lwip_kernel_offloads PRIVATE -Wall -Wextra -Werror)
endif()
