if(DIPLAY_FUZZ)
  if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_C_COMPILER_ID MATCHES "Clang" OR NOT DIPLAY_SANITIZERS)
    message(FATAL_ERROR "Fuzz targets require Linux Clang and DIPLAY_SANITIZERS=ON")
  endif()
  add_compile_options(-fsanitize=fuzzer-no-link)
  add_link_options(-fsanitize=fuzzer-no-link)
endif()
if(DIPLAY_SANITIZERS)
  if(WIN32 OR NOT CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    message(FATAL_ERROR "Sanitizers require a supported Linux compiler")
  endif()
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
  # lwIP pools remain one allocation from ASan's perspective. Guard individual pool/heap
  # allocations as well so an intra-pool overwrite is not hidden by that allocator model.
  add_compile_definitions(MEM_OVERFLOW_CHECK=2 MEMP_OVERFLOW_CHECK=2 MEM_SANITY_CHECK=1 MEMP_SANITY_CHECK=1)
endif()
